cmake_minimum_required(VERSION 3.16)
project(FlaxProtobufConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf CONFIG REQUIRED)
add_executable(protobuf-smoke main.cpp ../grpc-smoke/proto/echo.proto)
target_link_libraries(protobuf-smoke PRIVATE protobuf::libprotobuf)
target_include_directories(protobuf-smoke PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
protobuf_generate(TARGET protobuf-smoke IMPORT_DIRS ../grpc-smoke/proto APPEND_PATH)
enable_testing()
add_test(NAME protobuf-roundtrip COMMAND protobuf-smoke)
